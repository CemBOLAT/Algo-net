\section{Input Variables}

The following sets and parameters are defined as input data to the model:

\begin{itemize}
    \item $V$: Set of vertices (container locations).
    \item $T$: Set of container types.
    \item $r \in T$: Residential container type.
    \item $C_t$: Maximum number of residential vertices that can be assigned to a building of type $t \in T \setminus \{r\}$.
    \item $D_t$: Service distance threshold for type $t \in T \setminus \{r\}$, indicating the maximum distance within which the type can serve residential vertices. Distances follow a mixed Manhattan–Euclidean metric.
    \item $S_t$: Number of containers that constitute one complete group of type $t \in T \setminus \{r\}$.
    \item $\Delta_t$: Maximum allowable distance between any two vertices belonging to the same group of type $t \in T \setminus \{r\}$.
    \item $d(u,v)$: Mixed Manhattan–Euclidean distance between vertices $u,v \in V$.
    \item $h(u,v)$: Euclidean distance between vertices $u,v \in V$, used for compactness or clustering penalties.
\end{itemize}

\section{Precomputed Variables}

The following auxiliary sets are precomputed based on spatial relationships between vertices and type-specific parameters.

\vspace{1em}

\textbf{Candidate Subgroups:}

\[
U(t) = 
\Bigl\{
    U \subseteq V \ \Big| \ |U| = S_t
\Bigr\},
\quad \forall t \in T \setminus \{r\}.
\]

\textbf{Explanation:}  
For each non-residential type $t$, the set $U(t)$ contains all possible subsets of vertices of size $S_t$.  
Each subset represents a potential configuration (subgraph) of containers that could form one complete group of type $t$.

\vspace{1.5em}

\textbf{Feasible Groups Under Diameter Constraint:}

\[
G_t = 
\Bigl\{
    U \in U(t) \ \Big| \ 
    d(u,v) \le \Delta_t,\ 
    \forall u,v \in U
\Bigr\},
\quad \forall t \in T \setminus \{r\}.
\]

\textbf{Explanation:}  
The subset $G_t$ filters $U(t)$ by removing all candidate groups whose internal pairwise distances exceed the maximum allowable limit $\Delta_t$.  
Thus, $G_t$ includes only compact, spatially feasible groups for each non-residential type.

\vspace{1.5em}

\textbf{Neighborhood Groups for Residential Vertices:}

\[
S_{v,t} = 
\Bigl\{
    G \in G_t \ \Big| \ 
    \exists\, u \in G : d(u,v) \le D_t
\Bigr\},
\quad \forall v \in V, \ \forall t \in T \setminus \{r\}.
\]

\textbf{Explanation:}  
For each vertex $v$ and type $t$, the set $S_{v,t}$ contains all feasible groups of type $t$ that are within the service distance $D_t$ from $v$.  
Intuitively, it defines which type-$t$ facilities (groups) can potentially serve residential container $v$ under the mixed Manhattan–Euclidean distance metric.

\section{Decision Variables}

All decision variables are binary.

\vspace{1em}

\textbf{Type Assignment Variables:}

\[
x_{v,t} =
\begin{cases}
1, & 
\text{if vertex } v \\ 
  & \text{is assigned type } t, \\[4pt]
0, & \text{otherwise.}
\end{cases}
\qquad 
\forall v \in V,\ 
\forall t \in T.
\]

\textbf{Explanation:}  
The variable $x_{v,t}$ indicates whether vertex $v$ hosts a container of type $t$.  
Each vertex can take at most one type, including the residential type $r$.

\vspace{1.5em}

\textbf{Residential-to-Group Assignment Variables:}

\[
y_{v,g,t} =
\begin{cases}
1, & 
\text{if residential vertex } v \\ 
  & \text{is assigned to group } g \\ 
  & \text{of type } t, \\[4pt]
0, & \text{otherwise.}
\end{cases}
\qquad 
\forall v \in V,\ 
\forall g \in G_t,\ 
\forall t \in T \setminus \{r\}.
\]

\textbf{Explanation:}  
The variable $y_{v,g,t}$ links each residential vertex to a feasible group $g$ of a non-residential type $t$.  
It determines whether vertex $v$ is served by that particular facility group.

\vspace{1.5em}

\textbf{Group Activation Variables:}

\[
u_{g,t} =
\begin{cases}
1, & 
\text{if group } g \\ 
  & \text{of type } t \text{ is activated}, \\[4pt]
0, & \text{otherwise.}
\end{cases}
\qquad 
\forall g \in G_t,\ 
\forall t \in T \setminus \{r\}.
\]

\textbf{Explanation:}  
The binary variable $u_{g,t}$ activates or deactivates group $g$ of type $t$.  
Only activated groups can serve residential vertices and contribute to the model’s coverage and capacity constraints.

\section{Objective}

Maximize the total number of residential containers (where $r$ denotes the residential type):
\[
\max \sum_{v \in V} x_{v,r}
\]

\section{Core Constraints}

\begin{enumerate}
    \item \textbf{Single type per vertex:}
    \[
    \sum_{t \in T} x_{v,t} \le 1 
    \qquad \forall v \in V.
    \]
    \textbf{Explanation:}  
    Each vertex can host at most one container type, including the residential type $r$.

    \vspace{0.7em}


    \item \textbf{Neighborhood coverage (Rainbow constraint):}
    \[
    \sum_{g \in S_{v,t}} u_{g,t} \ge x_{v,r}
    \qquad 
    \forall v \in V,\ 
    \forall t \in T \setminus \{r\}.
    \]
    \textbf{Explanation:}  
    For every residential vertex $v$, and for each non-residential type $t$,  
    at least one activated group $u_{g,t}$ within $S_{v,t}$ must exist within service distance $D_t$.  
    This ensures that every residential vertex is covered by all required non-residential facilities.

    \vspace{0.7em}

    \item \textbf{Group size consistency:}
    \[
    u_{g,t} \le x_{v,t}
    \qquad 
    \forall t \in T \setminus \{r\},\ 
    \forall g \in G_t,\ 
    \forall v \in g.
    \]
    \textbf{Explanation:}  
    A group $g$ of type $t$ can only be activated if all its member vertices $v$ are assigned as that same type.  
    This enforces internal consistency between group activation and vertex-type assignment.

    \vspace{0.7em}

    \item \textbf{Group–vertex membership coherence:}
    \[
    x_{v,t} \le 
    \sum_{\substack{g \in G_t \\ v \in g}} u_{g,t},
    \qquad
    \sum_{\substack{g \in G_t \\ v \in g}} u_{g,t} \le 1,
    \qquad
    \forall v \in V,\ 
    \forall t \in T \setminus \{r\}.
    \]
    \textbf{Explanation:}  
    A vertex $v$ can be assigned to type $t$ only if it belongs to at least one active group of that type,  
    and each vertex can participate in at most one such group.  
    These two inequalities jointly prevent cross-group overlaps and maintain group exclusivity.

% --- Assignment equality constraint ---
\item \textbf{Assignment equality constraint:}
\[
\sum_{g \in S_{v,t}} y_{v,g,t} = x_{v,r},
\qquad
\forall v \in V,\ \forall t \in T \setminus \{r\}.
\]

\textbf{Explanation:}  
This equality enforces that every residential vertex $v$ must be assigned to exactly one
active group $g$ of each non-residential type $t$.
If $x_{v,r}=1$, the vertex must appear in one and only one assignment $y_{v,g,t}=1$.
If $x_{v,r}=0$, no assignment can exist for that vertex.
This ensures that each residential container receives full coverage
and activates the relevant demand and capacity constrain

% --- Demand (capacity) constraint ---
\item \textbf{Demand (capacity) constraint:}
\[
\sum_{v \in V} y_{v,g,t} \le C_t \, u_{g,t}
\qquad
\forall t \in T \setminus \{r\},\ \forall g \in G_t.
\]

\textbf{Explanation:}  
This constraint enforces the demand–capacity relationship between residential vertices and
non-residential groups.  
For each group $g$ of type $t$, the left-hand side represents the total number of residential vertices
assigned to that group.  
The right-hand side, $C_t \, u_{g,t}$, limits this number by the capacity parameter $C_t$.
If the group is inactive ($u_{g,t}=0$), no residential vertex can be assigned to it.
If the group is active ($u_{g,t}=1$), it may serve up to $C_t$ residential vertices but not more.
This ensures that the total service load of each non-residential group
never exceeds its predefined maximum capacity.

\vspace{1em}

% --- Linking constraints ---
\textbf{Linking constraints between assignments and activations:}
\[
y_{v,g,t} \le u_{g,t},
\qquad
y_{v,g,t} \le x_{v,r},
\qquad
\forall v \in V,\ \forall t \in T \setminus \{r\},\ \forall g \in S_{v,t}.
\]

\textbf{Explanation:}  
These linking constraints preserve logical consistency between different decision variables:
\begin{itemize}
    \item The inequality $y_{v,g,t} \le u_{g,t}$ ensures that a residential vertex $v$ can only be assigned
    to a group $g$ if that group is activated.
    \item The inequality $y_{v,g,t} \le x_{v,r}$ guarantees that only vertices designated as residential
    ($x_{v,r}=1$) can receive service from a non-residential group.
\end{itemize}
Together, these two relationships prevent infeasible assignments such as
linking inactive groups to residential vertices or assigning non-residential vertices
to any facility group.


\end{enumerate}
